	private Path currentPdf() {
		if (fixturePdf != null) {
			return fixturePdf;
		}

		return downloadedPdf;
	}

	private void deleteFixture() {

		if (fixturePdf == null) {
			return;
		}

		try {
			boolean deleted = Files.deleteIfExists(fixturePdf);

			if (deleted) {
				log.info("PDF test fixture deleted successfully: {}" + fixturePdf);
			} else {
				System.out.println("PDF test fixture does not exist: " + fixturePdf);
			}

		} catch (IOException e) {
			System.err.println("Failed to delete PDF test fixture: " + fixturePdf);
			e.printStackTrace();

		} finally {
			fixturePdf = null;
		}
	}

	private void createPdfFixture(String content) {
		try {
			Path directory = Path.of(System.getProperty("user.dir"), "target", "pdf-test-fixtures");

			Files.createDirectories(directory);

			fixturePdf = directory.resolve("test-fixture.pdf");

			try (PDDocument document = new PDDocument()) {
				PDPage page = new PDPage();
				document.addPage(page);

				if (content != null && !content.isEmpty()) {
					try (PDPageContentStream stream = new PDPageContentStream(document, page)) {

						stream.beginText();
						stream.setFont(new PDType1Font(Standard14Fonts.FontName.HELVETICA), 12);
						stream.newLineAtOffset(50, 700);
						stream.showText(content);
						stream.endText();
					}
				}

				document.save(fixturePdf.toFile());
			}

		} catch (IOException e) {
			throw new RuntimeException("Failed to create PDF test fixture", e);
		}
	}

	@Then("the downloaded file should have a PDF extension")
	public void the_downloaded_file_should_have_a_pdf_extension() {

		Path pdfPath = currentPdf();

		Assert.assertNotNull(pdfPath, "Downloaded PDF path should not be null");

		Assert.assertTrue(Files.exists(pdfPath), "Downloaded file does not exist: " + pdfPath);

		String fileName = pdfPath.getFileName().toString().toLowerCase(Locale.ROOT);

		Assert.assertTrue(fileName.endsWith(".pdf"),
				"Downloaded file should have a .pdf extension, but was: " + fileName);
	}

	@Then("the order PDF should contain {string}")
	public void the_order_pdf_should_contain(String expectedText) {

		Path pdfPath = currentPdf();

		Assert.assertNotNull(pdfPath, "PDF path should not be null");

		Assert.assertTrue(Files.exists(pdfPath), "PDF file does not exist: " + pdfPath);

		try (PDDocument document = Loader.loadPDF(pdfPath.toFile())) {

			PDFTextStripper pdfTextStripper = new PDFTextStripper();
			String actualText = pdfTextStripper.getText(document);

			log.info("Actual PDF text: {}",actualText);
			log.info("Expected PDF text: {}",expectedText);
			
			Assert.assertTrue(actualText.contains(expectedText),
					"Expected text was not found in the PDF: " + expectedText + "\nActual PDF text:\n" + actualText);

		} catch (IOException e) {
			Assert.fail("Failed to read PDF file: " + pdfPath, e);
		}
	}

	@Then("the downloaded file should have a valid PDF signature")
	public void the_downloaded_file_should_have_a_valid_pdf_signature() throws IOException {

		Path pdfPath = currentPdf();

		Assert.assertNotNull(pdfPath, "PDF path should not be null");

		Assert.assertTrue(Files.exists(pdfPath), "PDF file does not exist: " + pdfPath);

		Assert.assertTrue(Files.size(pdfPath) >= 5, "PDF file is too small to contain a PDF signature");

		byte[] expectedSignature = { '%', 'P', 'D', 'F', '-' };

		byte[] actualSignature = new byte[expectedSignature.length];

		try (var inputStream = Files.newInputStream(pdfPath)) {
			int bytesRead = inputStream.read(actualSignature);

			Assert.assertEquals(bytesRead, expectedSignature.length, "Could not read the complete PDF signature");
		}

		Assert.assertEquals(actualSignature, expectedSignature,
				"Invalid PDF signature. Expected %PDF- at the beginning of the file");
	}

	@Then("the downloaded file should be readable by a PDF parser")
	public void the_downloaded_file_should_be_readable_by_a_pdf_parser() throws IOException {

		Path pdfPath = currentPdf();

		Assert.assertNotNull(pdfPath, "PDF path should not be null");

		Assert.assertTrue(Files.exists(pdfPath), "PDF file does not exist: " + pdfPath);

		Assert.assertTrue(Files.size(pdfPath) > 0, "PDF file is empty: " + pdfPath);

		try (PDDocument document = Loader.loadPDF(pdfPath.toFile())) {

			Assert.assertFalse(document.isEncrypted(), "PDF should not be encrypted");

			Assert.assertTrue(document.getNumberOfPages() > 0, "PDF should contain at least one page");

		} catch (IOException e) {
			Assert.fail("Downloaded file could not be parsed as a valid PDF: " + pdfPath, e);
		}
	}

	@When("the PDF download is checked")
	public void pdfDownloadIsChecked() {
		downloadedPdf = null;

		// A .crdownload file is not a completed PDF.
		Path directory = Paths.get(pdfDownloadDirectory);

		try (Stream<Path> files = Files.list(directory)) {
			boolean completedPdfExists = files.anyMatch(
					path -> Files.isRegularFile(path) && path.getFileName().toString().toLowerCase().endsWith(".pdf"));

			if (completedPdfExists) {
				downloadedPdf = directory.resolve("order.pdf");
			}
		} catch (IOException e) {
			throw new AssertionError("Unable to inspect download directory", e);
		}
	}
