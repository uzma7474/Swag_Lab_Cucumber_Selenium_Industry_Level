1. First, check the PDF header

Add this to PdfDownloadUtils:

public static boolean isValidPdf(Path pdfPath) {

    if (pdfPath == null || !Files.exists(pdfPath)) {
        return false;
    }

    try {
        byte[] header = Files.readAllBytes(pdfPath);

        if (header.length < 5) {
            return false;
        }

        String pdfHeader = new String(header, 0, 5, StandardCharsets.US_ASCII);

        log.info("PDF header: {}", pdfHeader);

        return "%PDF-".equals(pdfHeader);

    } catch (IOException e) {
        log.error("Unable to validate PDF header", e);
        return false;
    }
}


Then validate:

public static Path waitForPdfDownload(int timeoutSeconds) {

    Path downloadDir = Paths.get(
        System.getProperty("user.dir"),
        "downloads",
        "pdf"
    );

    long endTime =
        System.currentTimeMillis() + (timeoutSeconds * 1000L);

    while (System.currentTimeMillis() < endTime) {

        try {

            // Check whether Chrome is still downloading
            try (Stream<Path> files = Files.list(downloadDir)) {

                boolean downloadInProgress = files.anyMatch(
                    path -> path.getFileName()
                               .toString()
                               .endsWith(".crdownload")
                );

                if (downloadInProgress) {
                    Thread.sleep(500);
                    continue;
                }
            }

            // Find PDF
            Optional<Path> pdfFile;

            try (Stream<Path> files = Files.list(downloadDir)) {

                pdfFile = files
                    .filter(path -> path.toString()
                        .toLowerCase()
                        .endsWith(".pdf"))
                    .filter(path -> {
                        try {
                            return Files.size(path) > 0;
                        } catch (IOException e) {
                            return false;
                        }
                    })
                    .findFirst();
            }

            if (pdfFile.isPresent()) {

                Path pdf = pdfFile.get();

                // Make sure file size is stable
                long size1 = Files.size(pdf);

                Thread.sleep(1000);

                long size2 = Files.size(pdf);

                if (size1 == size2 && size2 > 0) {

                    log.info(
                        "PDF download completed: {} ({} bytes)",
                        pdf,
                        size2
                    );

                    return pdf;
                }
            }

            Thread.sleep(500);

        } catch (Exception e) {
            log.warn("Waiting for PDF download: {}", e.getMessage());
        }
    }

    throw new RuntimeException(
        "PDF download did not complete within "
        + timeoutSeconds + " seconds"
    );
}

public static String extractPdfText(Path pdfPath) {

    try (PDDocument document = Loader.loadPDF(pdfPath.toFile())) {

        PDFTextStripper stripper = new PDFTextStripper();

        String text = stripper.getText(document);

        log.info("========== PDF CONTENT ==========");
        log.info(text);
        log.info("=================================");

        return text;

    } catch (IOException e) {

        log.error(
            "Unable to read PDF: {}",
            pdfPath,
            e
        );

        throw new RuntimeException(
            "Unable to read downloaded PDF",
            e
        );
    }
}
