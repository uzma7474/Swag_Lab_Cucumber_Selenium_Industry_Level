package assertions;

import org.testng.Assert;

import constants.UserConstants;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import pages.LoginPage;

/**
 * Assertion class for SauceDemo Login functionality.
 *
 * Responsibilities: - Validate Login page - Validate login error message -
 * Validate password masking - Validate Login button state
 *
 * This class does not perform Selenium actions.
 */
public class LoginAssertion {

	private static final Logger log = LoggerFactory.getLogger(LoginAssertion.class);

	private final LoginPage loginPage;

	/**
	 * Constructor.
	 *
	 * @param loginPage LoginPage instance
	 */
	public LoginAssertion(LoginPage loginPage) {

		if (loginPage == null) {

			log.error("LoginPage cannot be null");

			throw new IllegalArgumentException("LoginPage cannot be null");
		}

		this.loginPage = loginPage;

		log.debug("LoginAssertion initialized successfully");
	}

	// ============================================================
	// Login Page Assertions
	// ============================================================

	/**
	 * Verifies that Login page is displayed.
	 */
	public void verifyLoginPageDisplayed() {

		log.info("Assertion: Verifying Login page is displayed");

		try {

			boolean displayed = loginPage.isLoginPageDisplayed();

			Assert.assertTrue(displayed, "Login page should be displayed");

			log.info("Assertion passed: Login page is displayed");

		} catch (AssertionError e) {

			log.error("Assertion failed: Login page is not displayed", e);

			throw e;

		} catch (Exception e) {

			log.error("Error while verifying Login page", e);

			throw e;
		}
	}

	// ============================================================
	// Error Message Assertions
	// ============================================================

	/**
	 * Verifies that login error message is displayed.
	 */
	public void verifyErrorMessageDisplayed() {

		log.info("Assertion: Verifying login error message is displayed");

		try {

			boolean displayed = loginPage.isErrorMessageDisplayed();

			Assert.assertTrue(displayed, "Login error message should be displayed");

			log.info("Assertion passed: Login error message is displayed");

		} catch (AssertionError e) {

			log.error("Assertion failed: Login error message is not displayed", e);

			throw e;

		} catch (Exception e) {

			log.error("Error while verifying login error message", e);

			throw e;
		}
	}

	/**
	 * Verifies that login error message is not displayed.
	 */
	public void verifyErrorMessageNotDisplayed() {

		log.info("Assertion: Verifying login error message is not displayed");

		try {

			boolean displayed = loginPage.isErrorMessageDisplayed();

			Assert.assertFalse(displayed, "Login error message should not be displayed");

			log.info("Assertion passed: Login error message is not displayed");

		} catch (AssertionError e) {

			log.error("Assertion failed: Login error message is still displayed", e);

			throw e;

		} catch (Exception e) {

			log.error("Error while verifying login error message visibility", e);

			throw e;
		}
	}

	/**
	 * Verifies login error message contains expected text.
	 *
	 * @param expectedMessage expected error message text
	 */
	public void verifyErrorMessageContains(String expectedMessage) {

		log.info("Assertion: Verifying login error message");

		try {

			verifyErrorMessageDisplayed();

			String actualMessage = loginPage.getErrorMessage();

			log.debug("Expected error message: {}", expectedMessage);

			log.debug("Actual error message: {}", actualMessage);

			Assert.assertTrue(actualMessage.contains(expectedMessage), "Expected error message to contain: '"
					+ expectedMessage + "', but actual message was: '" + actualMessage + "'");

			log.info("Assertion passed: Login error message contains expected text");

		} catch (AssertionError e) {

			log.error("Assertion failed: Login error message does not contain expected text", e);

			throw e;

		} catch (Exception e) {

			log.error("Error while validating login error message", e);

			throw e;
		}
	}

	/**
	 * Verifies exact login error message.
	 *
	 * @param expectedMessage expected complete error message
	 */
	public void verifyErrorMessageEquals(String expectedMessage) {

		log.info("Assertion: Verifying exact login error message");

		try {

			verifyErrorMessageDisplayed();

			String actualMessage = loginPage.getErrorMessage();

			Assert.assertEquals(actualMessage, expectedMessage, "Login error message does not match");

			log.info("Assertion passed: Exact login error message matched");

		} catch (AssertionError e) {

			log.error("Assertion failed: Login error message mismatch", e);

			throw e;

		} catch (Exception e) {

			log.error("Error while validating exact login error message", e);

			throw e;
		}
	}

	// ============================================================
	// Password Assertions
	// ============================================================

	/**
	 * Verifies that the password field is masked.
	 */
	public void verifyPasswordIsMasked() {

		log.info("Assertion: Verifying password field is masked");

		try {

			boolean masked = loginPage.isPasswordMasked();

			Assert.assertTrue(masked, "Password field should be masked");

			log.info("Assertion passed: Password field is masked");

		} catch (AssertionError e) {

			log.error("Assertion failed: Password field is not masked", e);

			throw e;

		} catch (Exception e) {

			log.error("Error while verifying password masking", e);

			throw e;
		}
	}

	// ============================================================
	// Login Button Assertions
	// ============================================================

	/**
	 * Verifies that Login button is displayed.
	 */
	public void verifyLoginButtonDisplayed() {

		log.info("Assertion: Verifying Login button is displayed");

		try {

			boolean displayed = loginPage.isLoginButtonDisplayed();

			Assert.assertTrue(displayed, "Login button should be displayed");

			log.info("Assertion passed: Login button is displayed");

		} catch (AssertionError e) {

			log.error("Assertion failed: Login button is not displayed", e);

			throw e;
		}
	}

	public void verifyInvalidCredentialsError() {

		log.info("Verifying invalid credentials error message");

		String actualMessage = loginPage.getErrorMessage();

		Assert.assertEquals(actualMessage, UserConstants.INVALID_CREDENTIALS_MESSAGE,
				"Invalid credentials error message is incorrect");

		log.info("Invalid credentials error message verified successfully");
	}

	/**
	 * Verifies that Login button is enabled.
	 */
	public void verifyLoginButtonEnabled() {

		log.info("Assertion: Verifying Login button is enabled");

		try {

			boolean enabled = loginPage.isLoginButtonEnabled();

			Assert.assertTrue(enabled, "Login button should be enabled");

			log.info("Assertion passed: Login button is enabled");

		} catch (AssertionError e) {

			log.error("Assertion failed: Login button is not enabled", e);

			throw e;

		} catch (Exception e) {

			log.error("Error while verifying Login button state", e);

			throw e;
		}
	}
}
